\largesection{Research Interests}

% I am broadly interested in generative models, world modeling, and multimodal learning. 

Having started my research career in 3D vision, I expanded my focus during my M.S. to Generative AI and Unified Multimodal Models (UMMs), leading to four first-author papers at NeurIPS, ICLR, CVPR, and ECCV. \end{cvbullets}

\pubgroup{International Conference}

\pubitem{[C4]}
  {Transferability Between Understanding and Generation in Unified Multimodal Models}
  {\publinks{https://arxiv.org/abs/2607.04423}{https://cvlab-kaist.github.io/UMM_Transferability/}{https://github.com/cvlab-kaist/UMM_Transferability}}
  {\me{Jiwon Kang}\eqc, Heeji Yoon\eqc, Jaewoo Jung, Jaewon Min, Minkyeong Jeon, Biyeon Hwang, Sangwon Jung, Seungryong Kim\corr}
  {European Conference on Computer Vision (ECCV 2026).}
  {\item Investigated cross-task transferability between image understanding and generation in Unified Multimodal Models, revealing that its strength strongly depends on architecture and emerges most clearly with a fully shared Transformer backbone and unified visual encoder. \item Leveraged understanding-to-generation transfer as a practical post-training strategy to improve targeted generative capabilities, including counting, spatial relations, and text rendering, while mitigating the distribution shift and quality degradation caused by direct generation fine-tuning.}

\pubitem{[C3]}
  {\label{APPLE}APPLE: Attribute-Preserving Pseudo-Labeling for Diffusion-Based Face Swapping}
  {\publinks{https://arxiv.org/abs/2601.15288}{https://cvlab-kaist.github.io/APPLE/}{https://github.com/cvlab-kaist/APPLE}}
  {\me{Jiwon Kang}\eqc, Yeji Choi\eqc, JoungBin Lee, Wooseok Jang, Jinhyeok Choi, Taekeun Kang, Yongjae Park, Myungin Kim, Seungryong Kim\corr}
  {IEEE/CVF Conference on Computer Vision and Pattern Recognition (CVPR 2026).}
  {\item Identified attribute loss in diffusion-based face swapping caused by masked inpainting conditions, and designed an attribute-preserving teacher using conditional deblurring and attribute-aware inversion to retain both global and fine-grained target attributes. \item Constructed high-fidelity pseudo-triplets from the teacher to train a direct-editing student on clean attribute conditions, achieving state-of-the-art attribute preservation with competitive identity transfer and consistently lying on or near the Pareto frontier across datasets.}
  % {\small\color{linkblue}\href{https://arxiv.org/abs/2601.15288}{Paper}~\textbar~\href{https://cvlab-kaist.github.io/APPLE/}{Project Page}}

\pubitem{[C2]}
  {A Noise is Worth Diffusion Guidance}
  {\publinks{https://arxiv.org/abs/2412.03895}{https://cvlab-kaist.github.io/NoiseRefine/}{https://github.com/cvlab-kaist/NoiseRefine}}
  {Donghoon Ahn\eqc, \me{Jiwon Kang}\eqc, Sanghyun Lee, Jaewon Min, Wooseok Jang, Minjae Kim, Hyungwon Cho, Sayak Paul, SeonHwa Kim, Eunju Cha\corr, Kyong Hwan Jin\corr, Seungryong Kim\corr}
  {International Conference on Learning Representations (ICLR 2026).}
  {\item Revealed that high-quality guidance-free diffusion is largely determined by the initial noise, identifying small low-magnitude, low-frequency components that transform Gaussian noise into a favorable noise space for unguided denoising. \item Proposed NoiseRefine, which learns to refine the initial noise once before sampling, reproducing guidance-level generation quality without additional guidance passes while reducing inference cost and requiring only 50K text-image pairs for training.}

\pubitem{[C1]}
  {Where and How to Perturb: On the Design of Perturbation Guidance in Diffusion and Flow Models}
  {\publinks{https://arxiv.org/abs/2506.10978}{https://cvlab-kaist.github.io/HeadHunter/}{https://github.com/cvlab-kaist/HeadHunter}}
  {Donghoon Ahn\eqc, \me{Jiwon Kang}\eqc, Sanghyun Lee, Minjae Kim, Wooseok Jang, Jaewon Min, Sangwu Lee, Sayak Paul, Seungryong Kim\corr}
  {Conference on Neural Information Processing Systems (NeurIPS 2025).}
  {\item Conducted the first head-level analysis of attention perturbation guidance in Diffusion Transformers, revealing that individual attention heads exhibit specialized and compositional effects on visual concepts such as structure, lighting, style, and texture. \item Proposed HeadHunter to iteratively select attention heads for arbitrary user-defined objectives and SoftPAG to continuously control perturbation strength, enabling fine-grained quality and style guidance on SD3 and FLUX.1 with compact head subsets and negligible inference overhead.}

\pubgroup{Workshop}

\pubitem{[W1]}
  {Self-Evolving Neural Radiance Fields}
  {\publinks{https://arxiv.org/abs/2312.01003}{https://cvlab-kaist.github.io/SE-NeRF/}{https://github.com/cvlab-kaist/SE-NeRF}}
  {Jaewoo Jung\eqc, Jisang Han\eqc, \me{Jiwon Kang}\eqc, Seongchan Kim, Min-Seop Kwak, Seungryong Kim\corr}
  {ICCV 2025 Workshop on Wild3D: 3D Modeling, Reconstruction, and Generation in the Wild.}
  {\item Formulated few-shot NeRF as an iterative teacher-student self-training problem, progressively expanding supervision with pseudo-labeled novel views generated by the teacher and promoting each trained student to the next teacher. \item Introduced cross-view semantic consistency based ray reliability estimation and reliability-aware distillation, allowing existing NeRF models to learn more robust geometry from sparse views without external 3D priors or architectural modifications.}

\pubitem{[P1]}
  {RAIN-GS: Relaxing Accurate Initialization Constraint for 3D Gaussian Splatting}
  {\publinks{https://arxiv.org/abs/2403.09413}{https://cvlab-kaist.github.io/RAIN-GS/}{https://github.com/cvlab-kaist/RAIN-GS}}
  {Jaewoo Jung\eqc, Jisang Han\eqc, Honggyu An\eqc, \me{Jiwon Kang}\eqc, Seonghoon Park\eqc, Seungryong Kim\corr}
  {Under Review.}
  {\item Analyzed why 3D Gaussian Splatting critically depends on accurate SfM initialization, showing that limited Gaussian transport and missing low-frequency scene structure prevent robust coarse-to-fine optimization from random or noisy point clouds. \item Proposed RAIN-GS with Sparse-Large-Variance initialization, progressive Gaussian low-pass filtering, and Adaptive Bound-Expanding Split, enabling 3DGS to train from sub-optimal or random point clouds with performance comparable to or better than SfM initialization.}
